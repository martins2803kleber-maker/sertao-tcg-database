function placeResponseBytes(p){
  var available=(~p.flag)>>>0,spots=[];
  function addRange(base,controller,location,max){
    for(var i=0;i<max;i++)if(available&(1<<(base+i)))spots.push([controller,location,i]);
  }
  addRange(0,0,4,7);addRange(8,0,8,8);addRange(16,1,4,7);addRange(24,1,8,8);
  if(spots.length<p.count)throw Error('No legal field zone returned by OCGCore.');
  var out=new Uint8Array(p.count*3);
  for(var j=0;j<p.count;j++){out[j*3]=spots[j][0];out[j*3+1]=spots[j][1];out[j*3+2]=spots[j][2]}
  return out;
}

function paramVals(p){var a=p&0xffff,b=(p>>>16)&0xffff;return b&&b!==a?[a,b]:[a]}
function canExactParams(params,target){
  var sums=new Set([0]);
  params.forEach(function(p){var next=new Set();sums.forEach(function(s){paramVals(p).forEach(function(v){if(s+v<=target)next.add(s+v)})});sums=next});
  return sums.has(target);
}
function findSumSelection(p){
  var n=p.cards.length,must=p.must.map(function(x){return x.param});
  if(p.max){
    var min=Math.max(0,p.min),max=Math.min(p.max,n);
    function rec(start,left,chosen){
      if(left===0){return canExactParams(must.concat(chosen.map(function(i){return p.cards[i].param})),p.acc)?chosen.slice():null}
      for(var i=start;i<=n-left;i++){chosen.push(i);var r=rec(i+1,left-1,chosen);if(r)return r;chosen.pop()}
      return null;
    }
    for(var k=min;k<=max;k++){var r=rec(0,k,[]);if(r)return r}
  }else{
    var limit=Math.min(n,18);
    for(var mask=0;mask<(1<<limit);mask++){
      var chosen=[],params=must.slice();
      for(var i=0;i<limit;i++)if(mask&(1<<i)){chosen.push(i);params.push(p.cards[i].param)}
      if(chosen.length<p.min)continue;
      var minSum=0,maxSum=0,minSingle=0x7fffffff;
      params.forEach(function(pp){var vs=paramVals(pp),mn=Math.min.apply(null,vs),mx=Math.max.apply(null,vs);minSum+=mn;maxSum+=mx;minSingle=Math.min(minSingle,mn)});
      if(maxSum>=p.acc && minSum-minSingle<p.acc)return chosen;
    }
  }
  throw Error('No valid sum selection found.');
}

function handlePacket(pkt){
  if(!pkt.length)return null;
  var msg=pkt[0],dv=new DataView(pkt.buffer,pkt.byteOffset,pkt.byteLength);
  try{
    if(msg===2){log('<b>OCGCore:</b> invalid selection rejected; choose another legal command.');return {msg:2,player:0,retry:true}}
    if(msg===11)return parseIdle(pkt);
    if(msg===10)return parseBattle(pkt);
    if(msg===12)return {msg:12,player:dv.getUint8(1),cancelable:false};
    if(msg===13)return {msg:13,player:dv.getUint8(1),cancelable:false};
    if(msg===14)return parseOption(pkt);
    if(msg===15)return parseSelectCards(pkt,15);
    if(msg===16)return parseChain(pkt);
    if(msg===18||msg===24)return parsePlace(pkt,msg);
    if(msg===19)return {msg:19,player:dv.getUint8(1),code:u32(dv,2),positions:dv.getUint8(6)};
    if(msg===20)return parseSelectCards(pkt,20);
    if(msg===21||msg===25)return parseSort(pkt,msg);
    if(msg===22)return parseCounter(pkt);
    if(msg===23)return parseSum(pkt);
    if(msg===26)return parseSelectUnselect(pkt);
    if(msg===50){mirrorMovePacket(pkt)}
    else if(msg===90){
      var pl=dv.getUint8(1),n=u32(dv,2),o2=6,codes=[];
      for(var d=0;d<n&&o2+4<=pkt.length;d++){codes.push(u32(dv,o2));o2+=4}
      mirrorDraw(pl,codes);log((pl===0?'You':'BOT')+' drew '+n+' card(s).');
    }else if(msg===94){
      var pl2=dv.getUint8(1),lp=u32(dv,2);if(pl2===0)state.playerLP=lp;else state.botLP=lp;render();
    }else if(msg===91){
      var pl3=dv.getUint8(1),dam=u32(dv,2);if(pl3===0)state.playerLP=Math.max(0,state.playerLP-dam);else state.botLP=Math.max(0,state.botLP-dam);render();
    }else if(msg===92){
      var pl4=dv.getUint8(1),rec=u32(dv,2);if(pl4===0)state.playerLP+=rec;else state.botLP+=rec;render();
    }else if(msg===5){
      var winner=dv.getUint8(1);log('<b>'+(winner===0?'YOU WIN!':'BOT SERTAO WINS!')+'</b>');real.active=false;real.pending=null;updatePromptUI();
    }else if(msg===40){state.turn++;state.phase='DP';log('<b>Turn '+state.turn+'</b>')}
    else if(msg===41){
      var ph=dv.getUint16(1,true),map={1:'DP',2:'SP',4:'M1',8:'BP',16:'M2',32:'EP'};state.phase=map[ph]||state.phase;render();
    }
  }catch(e){console.error('packet '+msg,e);log('<b>UI packet error '+msg+':</b> '+String(e.message||e))}
  return null;
}

function presentPlayerPrompt(p){
  if(!p)return 'wait';
  if(p.msg===12){showYesNoPrompt(p,'ACTIVATE THIS EFFECT?');return 'wait'}
  if(p.msg===13){showYesNoPrompt(p,'CONFIRM');return 'wait'}
  if(p.msg===14){showOptionPrompt(p);return 'wait'}
  if(p.msg===15){showCardsPrompt(p,'SELECT CARD(S)');return 'wait'}
  /* IMPORTANT V15: automatic answers MUST NOT call pumpReal() recursively.
     While the current pump is active, the recursive call is ignored by
     real.pumping and the duel used to stop here forever. We answer and
     tell the same pump loop to continue instead. */
  if(p.msg===18||p.msg===24){real.mod.set_response_bytes(real.duel,placeResponseBytes(p));real.pending=null;return 'continue'}
  if(p.msg===19){showPositionPrompt(p);return 'wait'}
  if(p.msg===20){showCardsPrompt(p,'SELECT TRIBUTE(S)');return 'wait'}
  if(p.msg===21||p.msg===25){respondSort(p);real.pending=null;return 'continue'}
  if(p.msg===22){respondCounter(p);real.pending=null;return 'continue'}
  if(p.msg===23){respondIndices(findSumSelection(p));real.pending=null;return 'continue'}
  if(p.msg===26){showUnselectPrompt(p);return 'wait'}
  throw Error('Unsupported player prompt '+p.msg);
}

async function pumpReal(){
  if(!real.active)return;
  if(real.pumping)return;real.pumping=true;
  try{
    var safety=0;
    while(real.active&&safety++<240){
      if(safety%40===0)await new Promise(function(resolve){setTimeout(resolve,0)});
      var r=real.mod.process_duel(real.duel);
      var bytes=new Uint8Array(r.message||[]),packets=splitPackets(bytes),prompt=null,retry=false;
      packets.forEach(function(pkt){var p=handlePacket(pkt);if(p){if(p.retry)retry=true;else prompt=p}});
      if(retry){
        real.pending=null;
        if((real.retryCount=(real.retryCount||0)+1)>5)throw Error('OCGCore rejected repeated selections. Restart the duel.');
        continue;
      }
      real.retryCount=0;
      if(prompt){
        real.pending=prompt;updateRealButtons();
        if(prompt.player===1){await autoBotPrompt(prompt);continue}
        if(prompt.msg===16||prompt.msg===10||prompt.msg===11){
          updatePromptUI();return;
        }
        var promptFlow=presentPlayerPrompt(prompt);
        if(promptFlow==='continue'){updatePromptUI();continue}
        return;
      }
      updateRealButtons();
      if(Number(r.status)===0)return;
    }
    if(safety>=240)throw Error('Safety stop: core avançou demais sem solicitar decisão.');
  }catch(e){
    real.active=false;real.pending=null;closeChoice();updatePromptUI();
    log('<b>OCGCore stopped:</b> '+String(e.message||e));
    $('#realHelp').textContent='ERRO REAL: '+String(e.message||e);
    console.error(e);
  }finally{real.pumping=false}
}
