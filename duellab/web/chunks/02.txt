arr.push(pool[Math.floor(Math.random()*pool.length)]);

}

function draw(side, announce){
  var deck=side==='player'?state.playerDeck:state.botDeck;
  var hand=side==='player'?state.playerHand:state.botHand;
  if(!deck.length)return;
  var c=deck.shift();hand.push(c);
  if(announce!==false)log((side==='player'?'Você':'BOT')+' comprou uma carta.');
}

function bestMonster(hand){
  var choices=hand.map(function(c,i){return {c:c,i:i}}).filter(function(x){return isMonster(x.c)});
  choices.sort(function(a,b){return atk(b.c)-atk(a.c)});
  return choices[0]||null;
}

function summon(side){
  var hand=side==='player'?state.playerHand:state.botHand;
  var field=side==='player'?state.playerField:state.botField;
  if(field.length>=5)return false;
  var pick=bestMonster(hand); if(!pick)return false;
  hand.splice(pick.i,1);field.push(pick.c);
  log((side==='player'?'Você':'BOT')+' invocou <b>'+cardName(pick.c)+'</b> ('+atk(pick.c)+' ATK).');
  return true;
}

function attack(side){
  var mine=side==='player'?state.playerField:state.botField;
  var opp=side==='player'?state.botField:state.playerField;
  if(!mine.length)return false;
  var a=mine.slice().sort(function(x,y){return atk(y)-atk(x)})[0];
  if(opp.length){
    var d=opp.slice().sort(function(x,y){return atk(x)-atk(y)})[0];
    var diff=atk(a)-atk(d);
    log((side==='player'?'Você':'BOT')+' atacou <b>'+cardName(d)+'</b> com <b>'+cardName(a)+'</b>.');
    if(diff>0){
      var idx=opp.indexOf(d);opp.splice(idx,1);
      if(side==='player')state.botLP=Math.max(0,state.botLP-diff);else state.playerLP=Math.max(0,state.playerLP-diff);
      log('Monstro destruído. Dano: <b>'+diff+'</b>.');
    }else if(diff<0){
      if(side==='player')state.playerLP=Math.max(0,state.playerLP+diff);else state.botLP=Math.max(0,state.botLP+diff);
      var mi=mine.indexOf(a);mine.splice(mi,1);
      log('Atacante destruído. Dano: <b>'+Math.abs(diff)+'</b>.');
    }else{
      opp.splice(opp.indexOf(d),1);mine.splice(mine.indexOf(a),1);
      log('Os dois monstros foram destruídos.');
    }
  }else{
    var damage=atk(a);
    if(side==='player')state.botLP=Math.max(0,state.botLP-damage);else state.playerLP=Math.max(0,state.playerLP-damage);
    log((side==='player'?'Você':'BOT')+' atacou diretamente por <b>'+damage+'</b>.');
  }
  checkWin();render();return true;
}

function botTurn(){
  if(state.playerLP<=0||state.botLP<=0)return;

  state.playerTurn=false;
  state.normalUsed=false;
  state.turn++;
  log('<b>Turno '+state.turn+': BOT Sertão</b>');

  draw('bot');

  /* No modo demo, o BOT prioriza colocar um monstro no campo.
     Se não vier monstro na mão, compra cartas extras até 3 tentativas. */
  var summoned=summon('bot');
  var tries=0;
  while(!summoned && tries<3 && state.botDeck.length){
    draw('bot',false);
    tries++;
    summoned=summon('bot');
  }

  if(!summoned){
    log('BOT não encontrou um monstro para invocar neste turno.');
  }

  render();

  setTimeout(function(){
    if(state.botField.length){
      attack('bot');
    }else{
      log('BOT encerrou a Fase de Batalha sem monstros.');
    }

    setTimeout(function(){
      state.playerTurn=true;
      state.turn++;
      state.normalUsed=false;
      log('<b>Turno '+state.turn+': Você</b>');
      draw('player');
      render();
    },450);
  },550);
}

function checkWin(){
  if(state.botLP<=0){log('<b>Você venceu!</b>');toast('VOCÊ VENCEU')}
  if(state.playerLP<=0){log('<b>BOT Sertão venceu!</b>');toast('BOT SERTÃO VENCEU')}
}

  state.previewMap.set(key,c);

}
