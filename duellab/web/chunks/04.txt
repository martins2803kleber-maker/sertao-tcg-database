function u32(dv,o){return dv.getUint32(o,true)}
function i32(dv,o){return dv.getInt32(o,true)}
function u64num(dv,o){ return Number(dv.getBigUint64(o,true)); }
function readCardRef(dv,o,seq32){
  var x={code:u32(dv,o),controller:dv.getUint8(o+4),location:dv.getUint8(o+5)};
  if(seq32){x.sequence=u32(dv,o+6);x.next=o+10}else{x.sequence=dv.getUint8(o+6);x.next=o+7}
  return x;
}
function splitPackets(bytes){
  var out=[],pos=0;
  var dv=new DataView(bytes.buffer,bytes.byteOffset,bytes.byteLength);
  while(pos+4<=bytes.length){
    var size=u32(dv,pos);pos+=4;
    if(size<1||pos+size>bytes.length)break;
    out.push(bytes.slice(pos,pos+size));pos+=size;
  }
  return out;
}
function cardByCode(code){ return state.byId.get(String(code)) || real.coreById.get(Number(code)) || {id:code,name:'Card '+code,atk:0,type:'Monster'}; }
function mirrorRemoveHand(side,code){
  var hand=side===0?state.playerHand:state.botHand;
  var ix=hand.findIndex(function(c){return Number(c.id||c.baseId||c.originalId)===Number(code)});
  if(ix>=0)return hand.splice(ix,1)[0];
  return cardByCode(code);
}
function mirrorSummon(side,code){
  var field=side===0?state.playerField:state.botField;
  var card=mirrorRemoveHand(side,code);
  if(field.length<5 && !field.some(function(c){return c===card}))field.push(card);
  render();
}
function mirrorDraw(side,codes){
  var hand=side===0?state.playerHand:state.botHand;
  var deck=side===0?state.playerDeck:state.botDeck;
  codes.forEach(function(code){
    var ix=deck.findIndex(function(c){return Number(c.id||c.baseId||c.originalId)===Number(code)});
    var card=ix>=0?deck.splice(ix,1)[0]:cardByCode(code);
    hand.push(card);
  });
  render();
}

function locPool(side,loc){
  if(loc===1)return side===0?state.playerDeck:state.botDeck;
  if(loc===2)return side===0?state.playerHand:state.botHand;
  if(loc===4)return side===0?state.playerField:state.botField;
  if(loc===8)return side===0?state.playerSpellField:state.botSpellField;
  if(loc===16)return side===0?state.playerGY:state.botGY;
  if(loc===32)return side===0?state.playerBanished:state.botBanished;
  if(loc===64)return side===0?state.playerExtra:state.botExtra;
  return null;
}
function removeMirrorCard(side,loc,code,seq){
  var pool=locPool(side,loc);if(!pool)return null;
  if(loc===4||loc===8){
    if(seq!=null&&seq>=0&&seq<pool.length){
      var c=pool[seq];
      if(c&&(!code||Number(c.id||c.baseId||c.originalId)===Number(code))){pool[seq]=null;return c}
    }
    for(var z=0;z<pool.length;z++){var cc=pool[z];if(cc&&(!code||Number(cc.id||cc.baseId||cc.originalId)===Number(code))){pool[z]=null;return cc}}
    return null;
  }
  if(seq!=null&&seq>=0&&seq<pool.length){
    var candidate=pool[seq];
    if(candidate&&(!code||Number(candidate.id||candidate.baseId||candidate.originalId)===Number(code)))return pool.splice(seq,1)[0];
  }
  var ix=pool.findIndex(function(c){return c&&(!code||Number(c.id||c.baseId||c.originalId)===Number(code))});
  return ix>=0?pool.splice(ix,1)[0]:null;
}
function addMirrorCard(side,loc,code,seq,card){
  var pool=locPool(side,loc);if(!pool)return;
  card=card||cardByCode(code);
  if(loc===4||loc===8){
    var slot=(seq!=null&&seq>=0&&seq<5)?seq:pool.findIndex(function(x){return !x});
    if(slot<0)slot=0;pool[slot]=card;return;
  }
  if(seq!=null&&seq>=0&&seq<=pool.length)pool.splice(seq,0,card);else pool.push(card);
}
function mirrorMovePacket(pkt){
  var dv=new DataView(pkt.buffer,pkt.byteOffset,pkt.byteLength),o=1;
  var code=u32(dv,o);o+=4;
  function li(){var x={controller:dv.getUint8(o),location:dv.getUint8(o+1),sequence:u32(dv,o+2),position:u32(dv,o+6)};o+=10;return x}
  var prev=li(),cur=li();
  var moved=removeMirrorCard(prev.controller,prev.location,code,prev.sequence);
  addMirrorCard(cur.controller,cur.location,code,cur.sequence,moved||cardByCode(code));
  render();
}
