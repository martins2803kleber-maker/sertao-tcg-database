function mostlyJapanese(s){
  s=String(s||'');
  if(!s)return false;
  var jp=(s.match(/[\u3040-\u30ff\u3400-\u9fff]/g)||[]).length;
  return jp > Math.max(4, s.length * 0.16);
}
function englishCardDesc(c){
  var raw=c&&c.raw?c.raw:c;
  if(!raw)return '';
  var explicit=String(
    raw.desc_en || raw.description_en || raw.effect_en || raw.text_en ||
    raw.english_desc || ''
  ).trim();
  if(explicit)return explicit;

  var fallback=cardDesc(c);
  if(mostlyJapanese(fallback)){
    return 'English effect text is not available for this card in the current local database.';
  }
  return fallback;
}

async function loadDB(){
  for(var u of DB_URLS){
    try{
      var r=await fetch(u,{cache:'force-cache'});
      if(!r.ok)throw new Error('HTTP '+r.status);
      var data=await r.json();
      if(!Array.isArray(data))data=data.data||data.cards||[];
      state.db=data;
      state.byId=new Map();
      data.forEach(function(c){
        [c.id,c.baseId,c.originalId].filter(Boolean).forEach(function(id){state.byId.set(String(id),c)});
      });
      $('#dbPill').textContent='BANCO: '+data.length+' cartas';
      $('#dbPill').classList.add('ok');
      log('<b>Banco:</b> '+data.length+' cartas carregadas.');
      return;
    }catch(e){}
  }
  $('#dbPill').textContent='BANCO: falhou';
  log('<b>Erro:</b> não foi possível carregar yugioh.json.');
}

function parseYdkeMain(code){
  if(!code||code.indexOf('ydke://')!==0)return [];
  var parts=code.slice(7).split('!');
  function decode(s){
    if(!s)return [];
    try{
      var b=atob(s),out=[];
      for(var i=0;i+3<b.length;i+=4){
        out.push((b.charCodeAt(i)|(b.charCodeAt(i+1)<<8)|(b.charCodeAt(i+2)<<16)|(b.charCodeAt(i+3)<<24))>>>0);
      }
      return out;
    }catch(e){return []}
  }
  return decode(parts[0]||'');
}

function parseYdkeFull(code){
  var empty={main:[],extra:[],side:[]};
  if(!code||code.indexOf('ydke://')!==0)return empty;
  var parts=code.slice(7).split('!');
  while(parts.length<3)parts.push('');
  function decode(s){
    if(!s)return [];
    try{
      var b=atob(s),out=[];
      for(var i=0;i+3<b.length;i+=4){
        out.push((b.charCodeAt(i)|(b.charCodeAt(i+1)<<8)|(b.charCodeAt(i+2)<<16)|(b.charCodeAt(i+3)<<24))>>>0);
      }
      return out;
    }catch(e){return []}
  }
  return {main:decode(parts[0]),extra:decode(parts[1]),side:decode(parts[2])};
}
function parseYdke(code){ return parseYdkeFull(code).main; }
